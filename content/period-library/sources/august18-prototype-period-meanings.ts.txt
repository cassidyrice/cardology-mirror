/** Strip leading second-person phrasing so fragments embed cleanly in sentences. */
export function cleanSecondPerson(text: string): string {
  const stripped = text
    .trim()
    .replace(/^You're\s+/i, "")
    .replace(/^You are\s+/i, "")
    .replace(/^You’re\s+/i, "");
  return stripped.charAt(0).toUpperCase() + stripped.slice(1);
}

export function articleFor(word: string): string {
  return /^[aeiou]/i.test(word) ? "an" : "a";
}
